# 题目
给定一个正整数数列，和正整数p，设这个数列中的最大值是M，最小值是m，如果M <= m * p，则称这个数列是完美数列。     
现在给定参数p和一些正整数，请你从中选择尽可能多的数构成一个完美数列。   
<!-- more -->
输入格式：      
输入第一行给出两个正整数N和p，其中N（<= 10^5）是输入的正整数的个数，p（<= 10^9）是给定的参数。第二行给出N个正整数，每个数不超过10^9。  
输出格式：      
在一行中输出最多可以选择多少个数可以用它们组成一个完美数列。

输入样例：  
10 8    
2 3 20 4 5 1 6 7 8 9

输出样例：  
8

# 思路
首先将数列从小到大排序，设当前结果为max = 1，当前最长长度为temp = 1；从i = 0～n，j从i + max到n，【因为是为了找最大的max，所以下一次j只要从i的max个后面开始找】每次计算temp若大于max则更新max，最后输出max的值

# 注意
- m*p有可能超出int表示范围 用long long声明p
- 采用相当于滑动窗口的形式减少查找次数

# code
```C++
//错误示范 未考虑注意中的两点
#include <bits/stdc++.h>
using namespace std;
int main(int argc, const char * argv[]) {
    int n,p;
    cin>>n>>p;
    int a[n];
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
    }
    sort(a, a+n);
    int max = 1;
    for(int i=0;i<n;i++)
    {
        int temp = 1;
        for(int j=i+1;j<n;j++)
        {
            if(a[i]*p < a[j]) break;
            temp++;
            if(temp>max) max = temp;
        }
    }
    cout<<max<<endl;
    return 0;
}
```

```c++
//AC代码
#include <bits/stdc++.h>
using namespace std;
int main(int argc, const char * argv[]) {
    int n;
    long long p; //long long
    cin>>n>>p;
    int a[n];
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
    }
    sort(a, a+n);
    int max = 1,temp = 1; //temp不能放进单层循环内
    for(int i=0;i<n;i++)
    {
        for(int j=i+max;j<n;j++) //是max不是temp
        {
            if(a[i]*p < a[j]) break;
            temp++;
            if(temp>max) max = temp;
        }
    }
    cout<<max<<endl;
    return 0;
}
```
